	public static boolean isPrime(int number) {
		
		if(number <= 1) {
			System.out.println("Number is equal or less then 1");
			return false;
		}

		for(int i = 2 ; i < number ; i++) {
			if(number % i == 0)
				return false;
			else {
				return true;
			}
		}
		return false;
	}
